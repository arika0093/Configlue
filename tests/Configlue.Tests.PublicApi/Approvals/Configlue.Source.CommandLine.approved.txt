[assembly: System.Reflection.AssemblyMetadata("RepositoryUrl", "https://github.com/arika0093/Configlue")]
namespace Configlue.Source.CommandLine
{
    public sealed class CommandLineMappingBuilder
    {
        public CommandLineMappingBuilder() { }
        public void Map<TValue>(System.CommandLine.Argument<TValue> argument, string propertyPath) { }
        public void Map<TValue>(System.CommandLine.Option<TValue> option, string propertyPath) { }
        public void Map<TModel, TValue>(System.CommandLine.Argument<TValue> argument, System.Linq.Expressions.Expression<System.Func<TModel, TValue>> property) { }
        public void Map<TModel, TValue>(System.CommandLine.Option<TValue> option, System.Linq.Expressions.Expression<System.Func<TModel, TValue>> property) { }
        public void Map<TValue, TMember>(System.CommandLine.Argument<TValue> argument, string propertyPath, System.Func<TValue?, TMember?> convert) { }
        public void Map<TValue, TMember>(System.CommandLine.Option<TValue> option, string propertyPath, System.Func<TValue?, TMember?> convert) { }
        public void Map<TModel, TValue, TMember>(System.CommandLine.Argument<TValue> argument, System.Linq.Expressions.Expression<System.Func<TModel, TMember>> property, System.Func<TValue?, TMember?> convert) { }
        public void Map<TModel, TValue, TMember>(System.CommandLine.Option<TValue> option, System.Linq.Expressions.Expression<System.Func<TModel, TMember>> property, System.Func<TValue?, TMember?> convert) { }
    }
    public sealed class CommandLineSourceOptions
    {
        public CommandLineSourceOptions() { }
        public Configlue.Sources.StateFallbackCondition FallbackCondition { get; init; }
        public required string Id { get; init; }
        public required System.CommandLine.ParseResult ParseResult { get; init; }
        public int Priority { get; init; }
    }
    public static class CommandLineSourceRegistration
    {
        public static Configlue.ConfiglueSourceRegistration FromCommandLine(this Configlue.ConfiglueSourceSetBuilder sources, Configlue.Source.CommandLine.CommandLineSourceOptions options, System.Action<Configlue.Source.CommandLine.CommandLineMappingBuilder> configureMappings) { }
    }
    public static class CommonCommandLineSourcePreset
    {
        public static void UseCommonSources<TModel>(this Configlue.ConfiglueModelBuilder<TModel> model, Configlue.Source.Common.CommonSourceOptions options, System.CommandLine.ParseResult parseResult, System.Action<Configlue.Source.CommandLine.CommandLineMappingBuilder> configureMappings)
            where TModel : Configlue.IConfiglueFacadeModel<TModel> { }
    }
}
