[assembly: System.Reflection.AssemblyMetadata("RepositoryUrl", "https://github.com/arika0093/Configlue")]
namespace Configlue.Source.Common
{
    public sealed class CommonSourceOptions
    {
        public CommonSourceOptions() { }
        public required string ApplicationId { get; init; }
        public bool EnableEnvironment { get; init; }
        public bool EnableGlobalFile { get; init; }
        public bool EnableLocalFile { get; init; }
        public bool EnableSpecificFile { get; init; }
        public string? EnvironmentPrefix { get; init; }
        public System.Func<System.Collections.Generic.IEnumerable<System.Collections.Generic.KeyValuePair<string, string?>>>? EnvironmentVariables { get; init; }
        public Configlue.FileResourceOptions? FileResourceOptions { get; init; }
        public required string GlobalFileName { get; init; }
        public int GlobalWritePriority { get; init; }
        public string? LocalFilePath { get; init; }
        public int LocalWritePriority { get; init; }
        public System.Text.Json.JsonSerializerOptions? SerializerOptions { get; init; }
        public string? SpecificFilePath { get; init; }
        public int SpecificWritePriority { get; init; }
        public Configlue.Source.Common.CommonSourceWriteLayer WriteLayer { get; init; }
    }
    public static class CommonSourcePreset
    {
        public static void UseCommonSources<TModel>(this Configlue.ConfiglueModelBuilder<TModel> model, Configlue.Source.Common.CommonSourceOptions options)
            where TModel : Configlue.IConfiglueFacadeModel<TModel> { }
        public static void UseCommonSources<TModel>(this Configlue.ConfiglueModelBuilder<TModel> model, string applicationId, string? specificFilePath = null, string? environmentPrefix = null, Configlue.Source.Common.CommonSourceWriteLayer writeLayer = 0)
            where TModel : Configlue.IConfiglueFacadeModel<TModel> { }
    }
    public enum CommonSourceWriteLayer
    {
        Global = 0,
        Local = 1,
        Specific = 2,
        BestAvailable = 3,
    }
}
